  result=$(
    sed -r 's/\x1B\[[0-9;]*[mK]//g' "$f" \
      | tr '\r' '\n' \
      | grep 'Success rate:' \
      | tail -1
  )

  printf "%-32s %s\n" \
    "$(basename "$f" .log)" \
    "${result:-进行中/尚无完整 episode}"
done
